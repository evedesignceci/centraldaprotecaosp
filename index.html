// ========================================== // BACKEND API - CENTRAL DA PROTEÇÃO (GOOGLE APPS SCRIPT) // ========================================== function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName('Leads'); if (!sh) { sh = ss.insertSheet('Leads'); sh.appendRow([ 'Data e Hora', 'Tipo de Lead', 'Canal de Conversão', 'Assunto', 'Detalhes da Interação', 'Status do Lead', 'Corretor Responsável', 'Imóvel de Interesse', 'Data de Retorno', 'Próxima Ação' ]); } let p = {}; try { if (e && e.postData && e.postData.contents) { p = JSON.parse(e.postData.contents); } } catch (err) { p = (e && e.parameter) ? e.parameter : {}; } // Se vier vazio do postData, tenta pegar dos parâmetros normais da URL/Form if (!p.nome && e && e.parameter) { p = e.parameter; } if (p.website) return resp('ignorado'); // Isca anti-robô const limpa = (v, max) => { const s = String(v || '').slice(0, max); return /^[=+\-@]/.test(s) ? "'" + s : s; }; // Monta os detalhes capturando tudo o que vier do site (Nome, Telefone, Veículo, Uso, Região, etc.) const partesDetalhes = []; if (p.nome) partesDetalhes.push('Nome: ' + limpa(p.nome, 80)); if (p.telefone) partesDetalhes.push('WhatsApp: ' + limpa(p.telefone, 20)); if (p.tipo) partesDetalhes.push('Tipo: ' + limpa(p.tipo, 50)); if (p.veiculo) partesDetalhes.push('Veículo: ' + limpa(p.veiculo, 100)); if (p.uso) partesDetalhes.push('Uso: ' + limpa(p.uso, 100)); if (p.regiao) partesDetalhes.push('Região: ' + limpa(p.regiao, 100)); if (p.detalhes || p.mensagem) partesDetalhes.push(limpa(p.detalhes || p.mensagem, 500)); const detalhesStr = partesDetalhes.join(' | '); sh.appendRow([ Utilities.formatDate(new Date(), 'America/Sao_Paulo', 'yyyy-MM-dd HH:mm:ss'), limpa(p.tipoLead || p.tipo || 'Cotação Veicular', 60), limpa(p.canal || p.origem || 'Link na Bio / IA', 60), limpa(p.assunto || 'Nova Cotação', 100), detalhesStr, 'Novo', limpa(p.corretor, 60), limpa(p.imovel || p.veiculo, 300), '', '' ]); return resp('sucesso', 'Lead gravado com sucesso!'); } catch (error) { return resp('erro', error.toString()); } finally { try { lock.releaseLock(); } catch (x) {} } } function doGet(e) { return ContentService.createTextOutput(JSON.stringify({ status: 'ativo', mensagem: 'Webhook funcionando perfeitamente!' })) .setMimeType(ContentService.MimeType.JSON); } function resp(status, mensagem, data) { const corpo = { status: status, mensagem: mensagem || '' }; if (data) corpo.data = data; return ContentService.createTextOutput(JSON.stringify(corpo)) .setMimeType(ContentService.MimeType.JSON); }
